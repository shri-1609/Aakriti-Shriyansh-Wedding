const music = document.getElementById("music");
const openBtn = document.getElementById("openBtn");
const musicBtn = document.getElementById("musicBtn");

function updateMusicButton() {
  musicBtn.textContent = music.paused
    ? "🎵 Play Music"
    : "⏸ Pause Music";
}

openBtn.addEventListener("click", async () => {
  document.getElementById("invitation").scrollIntoView({
    behavior: "smooth"
  });

  try {
    await music.play();
  } catch (error) {
    // The guest can start music using the music button.
  }

  updateMusicButton();
});

musicBtn.addEventListener("click", async () => {
  if (music.paused) {
    try {
      await music.play();
    } catch (error) {
      alert("Please check that the music file is uploaded correctly.");
    }
  } else {
    music.pause();
  }

  updateMusicButton();
});

music.addEventListener("play", updateMusicButton);
music.addEventListener("pause", updateMusicButton);

// Wedding date and time in Indian Standard Time.
const weddingDate = new Date("2026-12-05T20:00:00+05:30");

function updateCountdown() {
  const now = Date.now();
  const diff = weddingDate.getTime() - now;
  const el = document.getElementById("countdown");

  if (diff <= 0) {
    el.textContent = "The celebration day is here! ❤️";
    return;
  }

  const days = Math.floor(diff / 86400000);
  const hours = Math.floor(diff / 3600000) % 24;
  const mins = Math.floor(diff / 60000) % 60;
  const secs = Math.floor(diff / 1000) % 60;

  el.textContent =
    `${days} Days · ${hours} Hours · ${mins} Min · ${secs} Sec`;
}

updateCountdown();
setInterval(updateCountdown, 1000);

// Subtle falling floral symbols on the opening screen.
const hero = document.getElementById("hero");

for (let i = 0; i < 12; i++) {
  const petal = document.createElement("span");
  petal.className = "petal";
  petal.textContent = "✿";
  petal.style.left = (Math.random() * 100) + "%";
  petal.style.animationDelay = (Math.random() * 8) + "s";
  petal.style.animationDuration = (7 + Math.random() * 6) + "s";
  petal.style.fontSize = (12 + Math.random() * 15) + "px";
  hero.appendChild(petal);
}
